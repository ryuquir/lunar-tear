cmake_minimum_required(VERSION 3.13)
# Target board: pico (RP2040, 27-pin buffered scheme) | pico2 (RP2350A, same wiring, 520K SRAM) | xiao_rp2040 (no bus: loader/audio role only)
set(FC_BOARD "pico" CACHE STRING "famicart board profile")
if(FC_BOARD STREQUAL "pico2")
  set(PICO_BOARD pico2 CACHE STRING "" FORCE)
  set(PICO_PLATFORM rp2350-arm-s CACHE STRING "" FORCE)
elseif(FC_BOARD STREQUAL "rp2350b")
  set(PICO_BOARD famicart_rp2350b CACHE STRING "" FORCE)
  set(PICO_BOARD_HEADER_DIRS ${CMAKE_CURRENT_LIST_DIR}/boards CACHE STRING "" FORCE)
  set(PICO_PLATFORM rp2350-arm-s CACHE STRING "" FORCE)
elseif(FC_BOARD STREQUAL "revd_prg" OR FC_BOARD STREQUAL "revd_chr")
  set(PICO_BOARD famicart_rp2350b CACHE STRING "" FORCE)
  set(PICO_BOARD_HEADER_DIRS ${CMAKE_CURRENT_LIST_DIR}/boards CACHE STRING "" FORCE)
  set(PICO_PLATFORM rp2350-arm-s CACHE STRING "" FORCE)
elseif(FC_BOARD STREQUAL "reve_cpu" OR FC_BOARD STREQUAL "reve_ppu")
  set(PICO_BOARD pico2 CACHE STRING "" FORCE)
  set(PICO_PLATFORM rp2350-arm-s CACHE STRING "" FORCE)
elseif(FC_BOARD STREQUAL "reve_x")
  set(PICO_BOARD seeed_xiao_rp2350 CACHE STRING "" FORCE)
  set(PICO_PLATFORM rp2350-arm-s CACHE STRING "" FORCE)
elseif(FC_BOARD STREQUAL "xiao_audio")
  set(PICO_BOARD seeed_xiao_rp2040 CACHE STRING "" FORCE)
elseif(FC_BOARD STREQUAL "xiao_rp2040")
  set(PICO_BOARD seeed_xiao_rp2040 CACHE STRING "" FORCE)
else()
  set(PICO_BOARD pico CACHE STRING "" FORCE)
endif()
include(pico_sdk_import.cmake)
project(famicart C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()

file(GLOB CORE_SRC src/core/*.c src/core/mappers/*.c src/core/audio/*.c)
set(REVE_COMMON src/core/reve/reve_link.c src/core/reve/romcache.c src/pico/reve_linkhw.c src/pico/reve_clock.c)
if(FC_BOARD STREQUAL "reve_cpu")
  add_executable(famicart ${CORE_SRC} ${REVE_COMMON} src/reve_cpu/main.c src/reve_cpu/usb_cmd_reve.c)
elseif(FC_BOARD STREQUAL "reve_ppu")
  add_executable(famicart ${CORE_SRC} ${REVE_COMMON} src/reve_ppu/main.c)
elseif(FC_BOARD STREQUAL "reve_x")
  add_executable(famicart ${CORE_SRC} ${REVE_COMMON} src/reve_x/main.c src/pico/audio_out.c)
elseif(FC_BOARD STREQUAL "xiao_audio")
  add_executable(famicart ${CORE_SRC} src/xiao_audio/main.c src/pico/audio_out.c src/pico/audio_link.c)
elseif(FC_BOARD STREQUAL "revd_prg")
  add_executable(famicart ${CORE_SRC} src/revd_prg/main.c src/revd_prg/usb_cmd_revd.c src/pico/bus_revd.c src/pico/bankout.c src/pico/maplink.c src/pico/norflash.c src/pico/audio_out.c)
elseif(FC_BOARD STREQUAL "revd_chr")
  add_executable(famicart ${CORE_SRC} src/revd_chr/main.c src/pico/maplink.c)
else()
  add_executable(famicart ${CORE_SRC} src/pico/main.c src/pico/bus.c src/pico/audio_out.c src/pico/romslot.c src/pico/usb_cmd.c src/pico/audio_link.c)
endif()
if(FC_BOARD MATCHES "^reve_")
  pico_generate_pio_header(famicart ${CMAKE_CURRENT_LIST_DIR}/src/pico/reve.pio)
elseif(FC_BOARD STREQUAL "rp2350b")
  pico_generate_pio_header(famicart ${CMAKE_CURRENT_LIST_DIR}/src/pico/fc_bus_direct.pio)
elseif(FC_BOARD STREQUAL "revd_prg")
  pico_generate_pio_header(famicart ${CMAKE_CURRENT_LIST_DIR}/src/pico/fc_bus_revd.pio)
  pico_generate_pio_header(famicart ${CMAKE_CURRENT_LIST_DIR}/src/pico/maplink.pio)
elseif(FC_BOARD STREQUAL "revd_chr")
  pico_generate_pio_header(famicart ${CMAKE_CURRENT_LIST_DIR}/src/pico/maplink.pio)
else()
  pico_generate_pio_header(famicart ${CMAKE_CURRENT_LIST_DIR}/src/pico/fc_bus.pio)
endif()
target_include_directories(famicart PRIVATE src src/core src/pico)
target_compile_definitions(famicart PRIVATE PICO_BUILD=1 FC_BOARD_${FC_BOARD}=1)
if(FC_BOARD MATCHES "^reve_")
  target_compile_definitions(famicart PRIVATE FC_ROM_POOL_SIZE=4 FC_RAM_POOL_SIZE=49152)        # PRG/CHR pools are owned by the rev E mains
elseif(FC_BOARD STREQUAL "pico2" OR FC_BOARD STREQUAL "rp2350b")
  target_compile_definitions(famicart PRIVATE FC_ROM_POOL_SIZE=450560 FC_RAM_POOL_SIZE=49152)   # 440K + 48K of 520K
elseif(FC_BOARD STREQUAL "revd_prg" OR FC_BOARD STREQUAL "revd_chr")
  target_compile_definitions(famicart PRIVATE FC_ROM_POOL_SIZE=0 FC_RAM_POOL_SIZE=16384 FC_VIRTUAL_ROM=1)   # ROM/RAM are external chips; ExRAM only (the core must see FC_VIRTUAL_ROM too)
elseif(FC_BOARD STREQUAL "xiao_audio")
  target_compile_definitions(famicart PRIVATE FC_ROM_POOL_SIZE=16384 FC_RAM_POOL_SIZE=8192)
else()
  target_compile_definitions(famicart PRIVATE FC_ROM_POOL_SIZE=184320 FC_RAM_POOL_SIZE=45056)   # 180K + 44K of 264K (synth state + 4K USB buffer)
endif()
target_compile_options(famicart PRIVATE -O2 -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(famicart pico_stdlib pico_multicore pico_bootrom hardware_pio hardware_dma hardware_pwm hardware_flash hardware_irq hardware_clocks hardware_watchdog hardware_uart)
if(FC_BOARD STREQUAL "reve_ppu" OR FC_BOARD STREQUAL "reve_x")
  pico_enable_stdio_usb(famicart 0)                   # the PPU and XIAO MCUs are driven over the link only
else()
  pico_enable_stdio_usb(famicart 1)
endif()
pico_enable_stdio_uart(famicart 0)
pico_add_extra_outputs(famicart)
